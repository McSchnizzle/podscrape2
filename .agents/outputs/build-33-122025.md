```typescript
// web_ui_hosted/app/api/settings/schema/route.ts
import { NextResponse } from 'next/server'
import { requireAuth } from '@/lib/auth-guard'

export const dynamic = 'force-dynamic'

const SETTINGS_SCHEMA: Record<string, Record<string, {
  type: string
  default: number | string | boolean
  min?: number | null
  max?: number | null
}>> = {
  // 13 categories with all settings...
}

const AI_MODELS = {
  openai: { /* 8 models */ },
  elevenlabs: { /* 7 models */ },
  whisper: { /* 1 model */ }
}

export async function GET() {
  const auth = await requireAuth()
  if (!auth.authorized) return auth.error!
  return NextResponse.json({ settings: SETTINGS_SCHEMA, ai_models: AI_MODELS })
}
```

## Root Cause Confirmation

The original implementation failed on Vercel because:
1. `spawn('python3', [...])` requires a Python runtime
2. Vercel serverless functions run on Node.js without Python
3. The path `.venv/bin/python3` does not exist in production

The fix removes all Python dependencies by porting the static configuration data to TypeScript constants.
