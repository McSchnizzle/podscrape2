## Code Changes Summary

### Before (Dual-State Pattern)
```typescript
// Two separate state variables
const [filters, setFilters] = useState({ q, status, sortBy, sortDir });
const [pendingFilters, setPendingFilters] = useState({ q, status, sortBy, sortDir });

// Handler updates pending only
const handleFilterChange = (key, value) => {
  setPendingFilters(prev => ({ ...prev, [key]: value }));
};

// Apply copies pending to active
const applyFilters = () => {
  setFilters(pendingFilters);
  loadEpisodesWithFilters(pendingFilters);
};

// Inputs use pendingFilters with onKeyDown for Enter
<input value={pendingFilters.q} onChange={...} onKeyDown={handleEnter} />
```

### After (Single-State Pattern)
```typescript
// Single state variable
const [filters, setFilters] = useState({ q, status, sortBy, sortDir });

// Handler updates filters directly
const handleFilterChange = (key, value) => {
  setFilters(prev => ({ ...prev, [key]: value }));
};

// Apply uses current filters
const applyFilters = () => {
  loadEpisodesWithFilters(filters);
};

// Inputs use filters (no onKeyDown needed - Apply button is only way)
<input value={filters.q} onChange={...} />
```
